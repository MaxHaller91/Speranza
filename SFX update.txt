**`playAssign`** — Gear Whirr (C)
```js
export function playAssign(ctx) {
  const o = ctx.createOscillator();
  const g = ctx.createGain();
  o.type = 'sawtooth';
  o.frequency.setValueAtTime(80, ctx.currentTime);
  o.frequency.exponentialRampToValueAtTime(120, ctx.currentTime + 0.1);
  g.gain.setValueAtTime(0.2, ctx.currentTime);
  g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.12);
  const filt = ctx.createBiquadFilter();
  filt.type = 'bandpass'; filt.frequency.value = 100; filt.Q.value = 2;
  o.connect(filt); filt.connect(g); g.connect(ctx.destination);
  o.start(); o.stop(ctx.currentTime + 0.15);
}
```

**`playDemolish`** — Crash / Crumble (A)
```js
export function playDemolish(ctx) {
  const buf = ctx.createBuffer(1, ctx.sampleRate * 0.5, ctx.sampleRate);
  const data = buf.getChannelData(0);
  for (let i = 0; i < data.length; i++) {
    const t = i / ctx.sampleRate;
    data[i] = Math.exp(-t * 8) * (Math.random() * 2 - 1) * 0.6;
  }
  const src = ctx.createBufferSource();
  src.buffer = buf;
  const filt = ctx.createBiquadFilter();
  filt.type = 'lowpass'; filt.frequency.value = 800;
  filt.frequency.exponentialRampToValueAtTime(200, ctx.currentTime + 0.4);
  src.connect(filt); filt.connect(ctx.destination);
  src.start();
}
```
